# 半平面交

> **用途：** 求若干有向直线左侧半平面的交，结果以交多边形顶点序列给出。
>
> **复杂度：** 排序 $O(n\log n)$，双端队列扫描 $O(n)$，总时间 $O(n\log n)$；空间 $O(n)$。
>
> **依赖：** `Point`、叉积运算、数乘/减法运算与浮点误差常量 `eps`。输入应先保证能形成模板所需的非退化、有界交集。

```cpp
struct Line {
    Point s,e;
    Line(Point ss = {0,0}, Point ee = {0,0}) {
        s = ss,e = ee;
    }

    double angel() {
        return atan2l((e-s).y,(e-s).x);
    }
};

Point crs(Line a, Line b) {
    double x = (a.e-b.e)*(b.e-b.s);
    double y = (a.e-a.s)*(b.e-b.s);
    double T = x/y;
    return a.e-(a.e-a.s)*T;
}

bool right(Line a, Line b, Line c) {
    Point x = crs(b,c);
    return (a.e-a.s)*(x-a.e)<0;
}

void solve() {
    vector<Line> line;
    sort(line.begin(),line.end(),[](Line a, Line b) {
        double A = a.angel(), B = b.angel();
        return fabsl(A-B) > eps ? A<B : (a.e-a.s)*(b.e-a.e) < 0;
    });
    
    int sz = line.size();

    vector<Line> q(sz+9);
    q[1] = line[0];
    int h = 1, t = 1;
    for (int i = 1; i < sz; i++) {
        if (fabs(line[i].angel()-line[i-1].angel()) < eps) continue;
        while (h < t && right(line[i],q[t],q[t-1])) t--;
        while (h < t && right(line[i],q[h],q[h+1])) h++;
        q[++t] = line[i];
    }
    while (h < t && right(q[h],q[t],q[t-1])) t--;

    vector<Point> res;
    for (int i = h; i <= t; i++) {
        int j = i == t ? h : i+1;
        res.push_back(crs(q[i],q[j]));
    }
}
```
