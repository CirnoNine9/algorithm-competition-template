# 点分治

> **用途：** 对无向树做重心分解，在每层只处理经过当前重心的路径。具体统计方式按题意补充到重心处理和子树遍历的位置。
>
> **复杂度：** 每层遍历当前连通块，总时间复杂度为 $O(n\log n)$，辅助数组和标记的空间复杂度为 $O(n)$；分治递归深度为 $O(\log n)$，子树遍历递归深度最坏为 $O(n)$。
>
> **注意：** `tr` 使用 `pii` 保存邻点和边权；无权树把边权设为 `1`。代码中的 `int` 随 `org.cpp` 为 `i64`，路径距离及统计中间值不能溢出；若题目中的乘加可能超出 `i64`，只在对应表达式处改用 `i128`。

~~~cpp
vector<vector<pii>> tr(n+9);
vector<int> vis(n+9), sz(n+9);

auto cal = [&](auto &&cal, int rt) -> void {
    auto getSz = [&](auto &&dfs, int u, int fa) -> void {
        sz[u] = 1;
        for (auto [v, w] : tr[u]) {
            if (vis[v] || v == fa) continue;
            dfs(dfs, v, u);
            sz[u] += sz[v];
        }
    };
    getSz(getSz, rt, 0);

    int sum = sz[rt];
    auto getRt = [&](auto &&dfs, int u, int fa) -> void {
        for (auto [v, w] : tr[u]) {
            if (vis[v] || v == fa || sz[v] * 2 < sum) continue;
            rt = v;
            dfs(dfs, v, u);
        }
    };
    getRt(getRt, rt, 0);

    int ans = 0;
    auto work = [&](int u, int fa, int d, int op) -> void {
        auto dfs = [&](auto &&dfs, int u, int fa, int d) -> void {
            for (auto [v, w] : tr[u]) {
                if (vis[v] || v == fa) continue;
                dfs(dfs, v, u, d + w);
            }
        };
        dfs(dfs, u, fa, d);
    };
    
    // 统计 rt 子树内部的贡献
    work(rt, 0, 0, 1);
    for (auto [v, w] : tr[rt]) {
        if (vis[v]) continue;
        // 容斥减去儿子子树内部贡献
        work(v, rt, w, -1);
    }

    vis[rt] = 1;
    for (auto [v, w] : tr[rt]) {
        if (vis[v]) continue;
        cal(cal, v);
    }
};
cal(cal, 1);
~~~
