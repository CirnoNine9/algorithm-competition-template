# 边双连通分量

> **用途：** 将无向图按桥分解为边双连通分量；同一分量内任意两点之间不存在必经边。允许自环和平行边，自环不会成为桥。
>
> **编号：** 点编号为 `0..n-1`，边按加入顺序从 `0` 开始编号。
>
> **依赖：** `pii` 为 `array<int,2>`。

| 操作/结构 | 含义 | 复杂度 | 备注 |
|---|---|---|---|
| `EBCC(n)` | 传入点数 `n`，构造空图 | $O(n)$ | 点数固定 |
| `addEdge(u,v)` | 传入端点 `u,v`，添加无向边并返回边号 | 摊还 $O(1)$ | 允许自环和平行边 |
| `build()` | 无传入和返回值，重新计算桥与边双连通分量 | $O(n+m)$ | 使用显式 DFS 栈；继续加边后可重建 |
| `edge[i]` | 维护第 `i` 条边的两个端点 | - | 端点类型为 `pii` |
| `bridge[i]` | 维护第 `i` 条边是否为桥 | - | `build()` 后有效；桥为 `1`，否则为 `0` |
| `bel[u]` | 维护点 `u` 所属的边双连通分量编号 | - | `build()` 后有效，每个点恰属一个分量 |
| `comp[i]` | 维护第 `i` 个边双连通分量的点集 | - | 包含孤立点形成的单点分量 |
| `bridgeTree()` | 无传入，返回以边双为点、桥为边的森林邻接表 | $O(n+m)$ | 要求先调用 `build()`；节点 `i` 对应 `comp[i]` |

```cpp
struct EBCC {
    int n, tim;
    vector<pii> edge;
    vector<vector<int>> e, comp;
    vector<int> dfn, low, bridge, bel;

    EBCC(int n_ = 0) : n(n_), e(n_) {}

    int addEdge(int u, int v) {
        int id = edge.size();
        edge.push_back({u, v});
        e[u].push_back(id), e[v].push_back(id);
        return id;
    }

    int other(int id, int u) const {
        return edge[id][0]^edge[id][1]^u;
    }

    void build() {
        tim = 0, comp.clear();
        dfn.assign(n, -1), low.resize(n), bridge.assign(edge.size(), 0);
        vector<int> it(n), pe(n, -1), call;
        for (int s = 0; s < n; s++) {
            if (dfn[s] != -1) continue;
            dfn[s] = low[s] = tim++, call.push_back(s);
            while (call.size()) {
                int u = call.back();
                if (it[u] < (int)e[u].size()) {
                    int id = e[u][it[u]++], v = other(id, u);
                    if (id == pe[u]) continue;
                    if (dfn[v] == -1) {
                        pe[v] = id, dfn[v] = low[v] = tim++;
                        call.push_back(v);
                    } else low[u] = min(low[u], dfn[v]);
                    continue;
                }
                call.pop_back();
                if (pe[u] == -1) continue;
                int p = other(pe[u], u);
                if (low[u] > dfn[p]) bridge[pe[u]] = 1;
                low[p] = min(low[p], low[u]);
            }
        }

        bel.assign(n, -1);
        for (int s = 0; s < n; s++) {
            if (bel[s] != -1) continue;
            int id = comp.size();
            comp.push_back({}), call.push_back(s), bel[s] = id;
            while (call.size()) {
                int u = call.back();
                call.pop_back(), comp[id].push_back(u);
                for (int i : e[u]) {
                    int v = other(i, u);
                    if (!bridge[i] && bel[v] == -1)
                        bel[v] = id, call.push_back(v);
                }
            }
        }
    }

    vector<vector<int>> bridgeTree() const {
        vector<vector<int>> g(comp.size());
        for (int i = 0; i < (int)edge.size(); i++) {
            if (!bridge[i]) continue;
            int u = bel[edge[i][0]], v = bel[edge[i][1]];
            g[u].push_back(v), g[v].push_back(u);
        }
        return g;
    }
};
```
