# 多点求值

> **用途：** 给定多项式 $f$ 与 $m$ 个点 $x_i$，同时计算全部 $f(x_i)$。乘积树保存在 NTT 域，向下分治时通过逐点乘法传递余式信息。
>
> **复杂度：** $O(n\log n+m\log^2m)$ 时间，$O(n+m\log m)$ 空间。
>
> **依赖：** 基础版或卡常版 NTT、多项式乘法与多项式乘法逆。
>
> **使用条件：** 模数为 $998244353$；设 $n=|f|$，要求 $1\le n\le 2^{22}$、$1\le m\le 2^{23}$。输入系数与求值点均应在 $[0,mod)$ 内。参数 `rt` 是与 NTT 一致的 $2^{23}$ 次单位根；卡常版取默认值 `31`，基础版传入 `15311432`。

令

$$
g(x)=\prod_{i=0}^{m-1}(x-x_i).
$$

先在 NTT 域自底向上构造乘积树，再由反转多项式的乘法逆得到根结点余式。向下递归时，将当前频域表示分别乘上另一侧子树的乘积，只保留高半部分，即可得到两侧的余式信息。

**常数参考（ms，中位数，$n=m$）：**

| NTT | $10^5$ | $2\times10^5$ | $5\times10^5$ | $10^6$ |
| --- | ---: | ---: | ---: | ---: |
| 基础版，`rt=15311432` | 294.518 | 622.990 | 1348.324 | 3499.812 |
| 卡常版，`rt=31` | 187.814 | 416.136 | 861.394 | 1967.186 |

```cpp
vector<int> multipointEval(const vector<int> &f, const vector<int> &x, int rt = 31) {
    int s = x.size();
    if (s == 1) {
        int y = 0;
        for (int i = f.size()-1; i >= 0; i--) y = (y*x[0]+f[i])%mod;
        return {y};
    }

    int n = 1;
    while (n < s) n <<= 1;
    bool basic = rt != 31;
    auto twice = [&](vector<int> &a) {
        int k = a.size(), z = rt, v = 1;
        for (int i = 23; i > __lg(k)+1; i--) z = z*z%mod;
        vector<int> b = a;
        ntt(b, k, -1);
        for (int i = 0; i < k; i++) {
            b[i] = b[i]*v%mod;
            v = v*z%mod;
        }
        ntt(b, k, 1);
        if (!basic) a.insert(a.end(), b.begin(), b.end());
        else {
            vector<int> c(2*k);
            for (int i = 0; i < k; i++) c[2*i] = a[i], c[2*i+1] = b[i];
            a.swap(c);
        }
    };

    vector<vector<int>> tr(2*n);
    for (int i = 0; i < n; i++) {
        int y = i < s ? (mod-x[i])%mod : 0;
        tr[i+n] = {(y+1)%mod, (y-1+mod)%mod};
    }
    for (int u = n-1; u; u--) {
        auto &a = tr[2*u], &b = tr[2*u+1];
        int k = a.size();
        tr[u].resize(k);
        for (int i = 0; i < k; i++) tr[u][i] = (a[i]*b[i]-1+mod)%mod;
        if (u == 1) continue;
        twice(tr[u]);
        for (int i = 0; i < 2*k; i++) {
            tr[u][i] += basic ? (i%2 ? mod-1 : 1) : (i < k ? 1 : mod-1);
            if (tr[u][i] >= mod) tr[u][i] -= mod;
        }
    }

    vector<int> g = tr[1];
    ntt(g, n, -1);
    g.push_back(1);
    reverse(g.begin(), g.end());
    g.resize(f.size());
    g = inv(g);
    reverse(g.begin(), g.end());
    g *= f;
    g.erase(g.begin(), g.begin()+f.size()-1);
    g.resize(n);

    vector<int> ans(n);
    auto dfs = [&](auto &&dfs, int u, vector<int> a) -> void {
        if (u >= n) {
            ans[u-n] = a[0];
            return;
        }
        int k = a.size();
        ntt(a, k, 1);
        for (int v = 2*u; v <= 2*u+1; v++) {
            vector<int> b = tr[v^1];
            for (int i = 0; i < k; i++) b[i] = b[i]*a[i]%mod;
            ntt(b, k, -1);
            dfs(dfs, v, vector<int>(b.begin()+k/2, b.end()));
        }
    };
    dfs(dfs, 1, g);
    ans.resize(s);
    return ans;
}
```
