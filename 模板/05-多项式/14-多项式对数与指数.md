# 多项式对数与指数

> **用途：** 形式幂级数运算：$\ln f=\int f'/f$，以及求满足 $\ln g=f$ 的 $g=\exp f$。

**常数参考（ms）：**

| 操作 | NTT | 版本 | $n=10^5$ | $n=2\times10^5$ | $n=5\times10^5$ | $n=10^6$ |
| --- | --- | --- | ---: | ---: | ---: | ---: |
| `ln` | 普通 NTT | `ln` | 40.465 | 92.085 | 218.098 | 541.231 |
| `ln` | 卡常 NTT | `ln` | 35.878 | 78.793 | 155.623 | 320.767 |
| `exp` | 普通 NTT | 普通版 | 106.973 | 235.114 | 564.397 | 1339.878 |
| `exp` | 普通 NTT | 卡常版 | 21.579 | 44.916 | 152.436 | 297.341 |
| `exp` | 卡常 NTT | 普通版 | 90.471 | 188.262 | 395.030 | 827.420 |
| `exp` | 卡常 NTT | 卡常版 | 16.233 | 34.372 | 99.422 | 215.577 |

| 操作/结构 | 含义 | 复杂度 | 备注 |
| --- | --- | --- | --- |
| `D(f)` / `I(f)` | 求导 / 积分 | $O(n)$ | 积分所需的 $1,\ldots,n$ 均应在模数下可逆 |
| `ln(f)` | 返回 $\ln f$ | $O(n\log n)$ | 要求 $f(0)=1$ |
| 基础版 `exp(f)` | 递归 Newton 迭代求 $\exp f$ | $O(n\log n)$ | 要求 $f(0)=0$ |
| 卡常版 `exp(f)` | DC16 分治递推求 $\exp f$ | $O(n\log^2 n)$ | 要求 $f(0)=0$；配合卡常 NTT |

**多项式对数：**

```cpp
vector<int> D(vector<int> f) { //求导
    int n = f.size();
    for (int i = 0; i < n-1; i++) {
        f[i] = (i+1)*f[i+1]%mod;
    }
    f.resize(n-1);
    return f;
}

vector<int> I(vector<int> f) {
    int n = f.size();
    f.resize(n+1);
    for (int i = n; i >= 1; i--) {
        f[i] = f[i-1]*inv(i)%mod;
    }
    f[0] = 0;
    return f;
}

vector<int> ln(vector<int> f) {
    auto invF = inv(f);
    auto res = D(f);
    res *= invF;
    res.resize(f.size());
    res = I(res);
    res.resize(f.size());
    return res;
}
```

**多项式指数：**

```cpp
vector<int> exp(vector<int> f) {
    if (f.size() == 1) return {1};
    int n = f.size();
    int m = (n+1)/2;
    auto h = f;
    h.resize(m);
    h = exp(h);
    h.resize(n);
    auto g = ln(h);
    for (int i = 0; i < n; i++) {
        f[i] = f[i]-g[i];
        if (f[i] < 0) f[i] += mod;
    }
    f[0] += 1;
    f *= h;
    f.resize(n);
    return f;
}
```

**卡常版多项式指数：** 由 $f'=fh'$ 得到 $if_i=\sum_{j=0}^{i-1}f_j(i-j)h_{i-j}$，用 DC16 分治与 NTT 批量计算跨块贡献，小块内直接递推。

```cpp
vector<int> exp(vector<int> h) {
    using u64 = unsigned long long;
    const int B = 16;
    int deg = h.size(), lim = 1 << (deg > 1 ? __lg(deg-1)+1 : 0);
    vector<int> f(lim), gw(2*lim), hw(2*lim), buf;
    h.resize(lim);
    for (int i = 1; i < deg; i++) h[i] = i*h[i]%mod;
    f[0] = 1;
    buf.reserve(max(2ll, lim/8));

    auto calc = [&](auto &&calc, int l, int r, int off) -> void {
        if (l >= deg) return;
        int S = r-l;
        if (S <= 64) {
            int rr = min(r, deg);
            for (int i = l; i < rr; i++) {
                u64 sum = 0, cnt = 0;
                for (int j = l; j < i; j++) {
                    sum += (u64)f[j]*h[i-j];
                    if (++cnt == 15) sum %= mod, cnt = 0;
                }
                f[i] += sum%mod;
                if (f[i] >= mod) f[i] -= mod;
                if (i) f[i] = f[i]*inv(i)%mod;
            }
            return;
        }

        int blen = S/B, k = 2*blen;
        int nb = min(B, (deg-l+blen-1)/blen);
        if (l == 0) {
            for (int q = 0; q+1 < nb; q++) {
                buf.resize(k);
                int st = q*blen;
                copy(h.begin()+st, h.begin()+st+k, buf.begin());
                ntt(buf, k, 1);
                copy(buf.begin(), buf.end(), hw.begin()+off+q*k);
            }
        }
        int childOff = off+k*(B-1);

        for (int i = 0; i < nb; i++) {
            int L = l+i*blen;
            if (i) {
                buf.resize(k);
                for (int t = 0; t < k; t++) {
                    u64 sum = 0;
                    for (int j = 0; j < i; j++) sum += (u64)gw[off+j*k+t]*hw[off+(i-j-1)*k+t];
                    buf[t] = sum%mod;
                }
                ntt(buf, k, -1);
                for (int t = 0; t < blen && L+t < deg; t++) {
                    f[L+t] += buf[blen+t];
                    if (f[L+t] >= mod) f[L+t] -= mod;
                }
            }
            calc(calc, L, L+blen, childOff);
            if (i+1 < nb) {
                buf.resize(k);
                int len = min(blen, deg-L);
                copy(f.begin()+L, f.begin()+L+len, buf.begin());
                fill(buf.begin()+len, buf.end(), 0);
                ntt(buf, k, 1);
                copy(buf.begin(), buf.end(), gw.begin()+off+i*k);
            }
        }
    };

    calc(calc, 0, lim, 0);
    f.resize(deg);
    return f;
}
```
